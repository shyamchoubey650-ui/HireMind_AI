from sqlalchemy import Column, Integer, String, DateTime, ForeignKey
from datetime import datetime
from .database import Base  # adjust import if your Base lives elsewhere


class ProctorSnapshot(Base):
    __tablename__ = "proctor_snapshots"

    id = Column(Integer, primary_key=True, index=True)
    submission_id = Column(Integer, ForeignKey("assessment_submissions.id"))
    candidate_id = Column(Integer, ForeignKey("users.id"))
    question_id = Column(String, default="")
    file_path = Column(String, nullable=False)
    captured_at = Column(DateTime, default=datetime.utcnow)
